using System.Threading.Channels;
namespace Lista
{
    internal class Program
    {
        static void Main(string[] args)

        {
            List<int> Lista = new List<int>();
            List<int> ListaPar = new List<int>();
            List<int> ListaImpar = new List<int>();

            while (true)
            {
                Console.WriteLine("Digite números para a lista (Envie 0 para parar)");
                int NumeroAdicionar = Convert.ToInt32(Console.ReadLine());
                Lista.Add(NumeroAdicionar);
                if (NumeroAdicionar == 0)
                    break;
            }
            foreach (var item in Lista)

            {
                if (item % 2 == 0)
                    ListaPar.Add(item);
                else ListaImpar.Add(item);

            }
            Lista.Order();
            Console.WriteLine("A sua lista de números em ordem crescente é de: " + string.Join(", ", Lista));
            int SomaLista = Lista.Sum();
            Console.WriteLine("A soma dos números inseridos é de:\n " + SomaLista);

        }

    }
